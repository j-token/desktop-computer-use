cmake_minimum_required(VERSION 3.22)
project(dcu_drag_fixture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_executable(dcu-drag-fixture drag-fixture.cpp)
if(WIN32)
  target_compile_definitions(dcu-drag-fixture PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(dcu-drag-fixture PRIVATE user32 gdi32)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
  target_link_libraries(dcu-drag-fixture PRIVATE PkgConfig::GTK)
endif()
