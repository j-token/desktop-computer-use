cmake_minimum_required(VERSION 3.22)
project(desktop_computer_use VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
# Prefer a system package so offline builds work once dependencies are installed.
find_package(nlohmann_json 3.10 QUIET)
if(NOT nlohmann_json_FOUND)
  FetchContent_MakeAvailable(json)
endif()
add_library(dcu_contract INTERFACE)
target_include_directories(dcu_contract INTERFACE native/include)
target_link_libraries(dcu_contract INTERFACE nlohmann_json::nlohmann_json)
option(DCU_BUILD_BACKEND "Build the platform daemon" ON)
if(DCU_BUILD_BACKEND)
  if(WIN32)
    add_subdirectory(native/windows)
  else()
    add_subdirectory(native/linux)
  endif()

  add_executable(desktop-computer-use-native
    native/common/main.cpp
    native/common/server.cpp
    native/common/dispatch.cpp
    native/common/request_validation.cpp)
  target_link_libraries(desktop-computer-use-native PRIVATE dcu_contract dcu_platform)
  target_include_directories(desktop-computer-use-native PRIVATE native/common)
  if(WIN32)
    target_compile_definitions(desktop-computer-use-native PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(desktop-computer-use-native PRIVATE advapi32)
  else()
    find_package(Threads REQUIRED)
    target_link_libraries(desktop-computer-use-native PRIVATE Threads::Threads)
  endif()
endif()
option(BUILD_TESTING "Build native contract tests" ON)
if(BUILD_TESTING)
  enable_testing()
  add_executable(dcu-contract-tests
    native/common/dispatch_tests.cpp
    native/common/dispatch.cpp
    native/common/request_validation.cpp)
  target_link_libraries(dcu-contract-tests PRIVATE dcu_contract)
  if(WIN32)
    target_link_libraries(dcu-contract-tests PRIVATE advapi32)
  endif()
  add_test(NAME contract COMMAND dcu-contract-tests)
  if(UNIX AND NOT APPLE)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
      pkg_check_modules(DCU_TEST_GIO QUIET IMPORTED_TARGET gio-unix-2.0)
      if(DCU_TEST_GIO_FOUND)
        add_executable(dcu-portal-request-tests tests/integration/portal-request-tests.cpp)
        target_include_directories(dcu-portal-request-tests PRIVATE native/linux)
        target_link_libraries(dcu-portal-request-tests PRIVATE dcu_contract PkgConfig::DCU_TEST_GIO)
        add_test(NAME portal-request-lifetime COMMAND dcu-portal-request-tests)
        set_tests_properties(portal-request-lifetime PROPERTIES TIMEOUT 15)
      endif()
    endif()
  endif()
endif()
if(DCU_BUILD_BACKEND)
  install(TARGETS desktop-computer-use-native RUNTIME DESTINATION bin)
endif()
